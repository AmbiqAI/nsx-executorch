cmake_minimum_required(VERSION 3.24)
project(nsx_executorch_configure_smoke_ns LANGUAGES C CXX)

add_library(nsx_core INTERFACE)
add_library(nsx::core ALIAS nsx_core)
add_library(nsx_cmsis_core INTERFACE)
add_library(nsx::cmsis_core ALIAS nsx_cmsis_core)
add_library(nsx_soc_hal INTERFACE)
add_library(nsx::soc_hal ALIAS nsx_soc_hal)
add_library(nsx_board_flags INTERFACE)
add_library(nsx::board_flags ALIAS nsx_board_flags)
include(GNUInstallDirs)

set(NSX_EXECUTORCH_CMSIS_NN_PROVIDER ns CACHE STRING "" FORCE)
set(NSX_EXECUTORCH_ENABLE_PROFILING ON CACHE BOOL "" FORCE)
get_filename_component(_ns_cmsis_nn_fixture_root
  "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/ns-cmsis-nn" ABSOLUTE)
set(NSX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(NSX_APP_MODULE_DIR_nsx_cmsis_nn "fixtures/ns-cmsis-nn/nsx")
add_subdirectory("${_ns_cmsis_nn_fixture_root}/nsx" ns-cmsis-nn-provider)
install(TARGETS nsx_cmsis_nn EXPORT nsx_executorch_smoke_ns_targets)
install(EXPORT nsx_executorch_smoke_ns_targets
        FILE NsxExecutorchSmokeNsTargets.cmake
        DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/NsxExecutorchSmokeNs")
add_subdirectory("../.." nsx-executorch)

if(NOT TARGET nsx::executorch)
  message(FATAL_ERROR "nsx::executorch target was not created")
endif()
if(NOT EXECUTORCH_ENABLE_EVENT_TRACER)
  message(FATAL_ERROR "profiling did not enable stock ExecuTorch EventTracer")
endif()
if(NOT CMSIS_NN_LOCAL_PATH MATCHES "cmsis-nn-provider$")
  message(FATAL_ERROR "ns-provider adapter was not selected")
endif()
if(NOT EXISTS "${CMSIS_NN_LOCAL_PATH}/Include/arm_nnfunctions.h")
  message(FATAL_ERROR "ns provider headers were not materialized")
endif()
if(NOT TARGET nsx::cmsis_nn OR TARGET nsx::arm_cmsis_nn)
  message(FATAL_ERROR
    "ns provider target selection is incorrect")
endif()
if(NOT TARGET cmsis-nn)
  message(FATAL_ERROR "ns-cmsis-nn bridge did not create the cmsis-nn target")
endif()
get_target_property(_cmsis_nn_type cmsis-nn TYPE)
if(NOT _cmsis_nn_type STREQUAL "INTERFACE_LIBRARY")
  message(FATAL_ERROR "preconfigured ns provider must use an interface bridge")
endif()
if(NOT FETCHCONTENT_FULLY_DISCONNECTED)
  message(FATAL_ERROR "CMake dependency fetching was not disabled")
endif()
